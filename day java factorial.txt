import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        long fact=1;
        for(int i=1; i<=n; i++){
            fact=fact*i;
           }
        
               
        System.out.println("factorial="+fact);
        
}
}
