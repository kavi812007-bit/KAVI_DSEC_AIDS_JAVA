    public static void main(String[] args) {
        int a[] = {1, 2, 3, 4, 6, 7, 8, 9, 10};

        int n = 10;
        int sum = n * (n + 1) / 2;
        int actualSum = 0;

        for (int i = 0; i < a.length; i++) {
            actualSum = actualSum + a[i];
        }

        int missing = sum - actualSum;

        System.out.println("Missing number: " + missing);
    }
