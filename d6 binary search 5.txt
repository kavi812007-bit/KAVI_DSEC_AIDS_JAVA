import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int a[] = {10,20,30,40,50,60};
        int key=40;
        int pos=0;
        int low=0;
        int high=a.length-1;
        pos = -1;
        while(low<=high){
            int mid=(low+high)/2;
            if(a[mid]==key){
                pos=mid;
                break;
            }else if(key>a[mid]){
                low=mid+1;
            }else{
                high=mid-1;
            }
        }
        if(pos!=-1)
            System.out.println("found at index+pos");
            else
            System.out.println("not found");
    }
}
